with:

template <typename T, unsigned N1, unsigned N2>
constexpr std::array<T, N1 + N2> operator+(std::array<T, N1> lhs, std::array<T, N2> rhs)
{
    std::array<T, N1 + N2> result{};

    for (unsigned i = 0; i < N1; ++i)
        result[i] = lhs[i];

    for (unsigned i = 0; i < N2; ++i)
        result[N1 + i] = rhs[i];

    return result;
}
